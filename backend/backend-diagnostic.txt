generator client {

  provider = "postgresql"

  schemas  = ["public", "auth"]

// Ignore the Supabase Auth schema's models so Prisma won't try to manage them
model User {
  id String @id @db.Uuid
  @@schema("auth")
  @@ignore

  ACTIVE
  SUSPENDED
  @@schema("public")
}

  STAFF
  HOSPITAL_ADMIN
  PRIVACY_OFFICER
  SECURITY_INVESTIGATOR
  SUPER_ADMIN
  @@schema("public")
}

  SCHEDULED
  COMPLETED
  CANCELLED
  NO_SHOW
  @@schema("public")
}

enum ConsentStatus {
  ACTIVE
  REVOKED
  EXPIRED
  @@schema("public")
}

enum AccessRequestStatus {
  PENDING
  APPROVED
  REJECTED
  EXPIRED
  @@schema("public")
}

  TREATMENT
  DIAGNOSIS
  LAB_REVIEW
  PHARMACY_DISPENSE
  BILLING
  RESEARCH
  TEACHING
  AUDIT_REVIEW
  @@schema("public")

model Tenant {
  id        String       @id @default(dbgenerated("gen_random_uuid()")) @db.Uuid
  slug      String       @unique
  name      String
  status    TenantStatus @default(ACTIVE)
  createdAt DateTime     @default(now())
  updatedAt DateTime     @updatedAt

  memberships    Membership[]
  patients       Patient[]
  departments    Department[]
  appointments   Appointment[]
  consents       Consent[]
  accessRequests AccessRequest[]
  accessGrants   AccessGrant[]

  @@map("tenants")
  @@schema("public")
}

model UserIdentity {
  id          String   @id @default(dbgenerated("gen_random_uuid()")) @db.Uuid
  email       String   @unique
  displayName String?
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  memberships Membership[]

  @@map("user_identities")
  @@schema("public")
}

model Department {
  id        String   @id @default(dbgenerated("gen_random_uuid()")) @db.Uuid
  tenantId  String   @db.Uuid
  name      String
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  tenant      Tenant       @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  memberships Membership[]

  @@unique([tenantId, name])
  @@index([tenantId])
  @@map("departments")
  @@schema("public")
}

model Membership {
  id               String         @id @default(dbgenerated("gen_random_uuid()")) @db.Uuid
  tenantId         String         @db.Uuid
  userId           String         @db.Uuid
  role             MembershipRole
  active           Boolean        @default(true)
  departmentId     String?        @db.Uuid
  employmentStatus String         @default("ACTIVE")
  createdAt        DateTime       @default(now())
  updatedAt        DateTime       @updatedAt

  tenant     Tenant       @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  user       UserIdentity @relation(fields: [userId], references: [id], onDelete: Cascade)
  department Department?  @relation(fields: [departmentId], references: [id], onDelete: SetNull)

  appointments              Appointment[]
  accessRequestsAsRequester AccessRequest[] @relation("AccessRequestsAsRequester")
  accessGrantsAsGrantee     AccessGrant[]   @relation("AccessGrantsAsGrantee")

  @@unique([tenantId, userId])
  @@index([tenantId])
  @@index([departmentId])
  @@map("memberships")
  @@schema("public")
}

model Patient {
  id             String   @id @default(dbgenerated("gen_random_uuid()")) @db.Uuid
  tenantId       String   @db.Uuid
  firstName      String
  lastName       String
  dateOfBirth    DateTime @db.Date
  hospitalNumber String   @unique
  createdAt      DateTime @default(now())
  updatedAt      DateTime @updatedAt

  tenant         Tenant          @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  appointments   Appointment[]
  consents       Consent[]
  accessRequests AccessRequest[]
  accessGrants   AccessGrant[]

  @@index([tenantId])
  @@map("patients")
  @@schema("public")
}

model Appointment {
  id           String            @id @default(dbgenerated("gen_random_uuid()")) @db.Uuid
  tenantId     String            @db.Uuid
  patientId    String            @db.Uuid
  staffId      String            @db.Uuid
  scheduledAt  DateTime
  durationMins Int               @default(30)
  reason       String?
  status       AppointmentStatus @default(SCHEDULED)
  createdAt    DateTime          @default(now())
  updatedAt    DateTime          @updatedAt

  tenant  Tenant     @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  patient Patient    @relation(fields: [patientId], references: [id], onDelete: Cascade)
  staff   Membership @relation(fields: [staffId], references: [id], onDelete: Cascade)

  @@index([tenantId, scheduledAt])
  @@index([patientId])
  @@index([staffId])
  @@map("tb_appointments")
  @@schema("public")
}

model Consent {
  id               String        @id @default(dbgenerated("gen_random_uuid()")) @db.Uuid
  tenantId         String        @db.Uuid
  patientId        String        @db.Uuid
  purpose          String
  noticeVersion    String
  language         String        @default("en")
  grantedAt        DateTime      @default(now())
  effectiveFrom    DateTime      @default(now())
  expiresAt        DateTime?
  status           ConsentStatus @default(ACTIVE)
  revokedAt        DateTime?
  revocationReason String?
  capturedBy       String?       @db.Uuid
  createdAt        DateTime      @default(now())
  updatedAt        DateTime      @updatedAt

  tenant  Tenant  @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  patient Patient @relation(fields: [patientId], references: [id], onDelete: Cascade)

  @@index([tenantId, status])
  @@index([patientId])
  @@map("tb_consents")
  @@schema("public")
}

model AccessRequest {
  id                     String              @id @default(dbgenerated("gen_random_uuid()")) @db.Uuid
  tenantId               String              @db.Uuid
  requesterId            String              @db.Uuid
  patientId              String              @db.Uuid
  purpose                AccessPurpose
  reason                 String
  requestedDurationHours Int                 @default(4)
  status                 AccessRequestStatus @default(PENDING)
  decisionReason         String?
  decidedBy              String?             @db.Uuid
  decidedAt              DateTime?
  createdAt              DateTime            @default(now())
  updatedAt              DateTime            @updatedAt

  tenant    Tenant     @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  requester Membership @relation("AccessRequestsAsRequester", fields: [requesterId], references: [id], onDelete: Cascade)
  patient   Patient    @relation(fields: [patientId], references: [id], onDelete: Cascade)
  grant     AccessGrant?

  @@index([tenantId, status])
  @@index([requesterId])
  @@index([patientId])
  @@map("tb_access_requests")
  @@schema("public")
}

model AccessGrant {
  id               String        @id @default(dbgenerated("gen_random_uuid()")) @db.Uuid
  tenantId         String        @db.Uuid
  requestId        String        @unique @db.Uuid
  granteeId        String        @db.Uuid
  patientId        String        @db.Uuid
  purpose          AccessPurpose
  grantedAt        DateTime      @default(now())
  expiresAt        DateTime
  revokedAt        DateTime?
  revocationReason String?
  createdAt        DateTime      @default(now())
  updatedAt        DateTime      @updatedAt

  tenant  Tenant        @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  request AccessRequest @relation(fields: [requestId], references: [id], onDelete: Cascade)
  patient Patient       @relation(fields: [patientId], references: [id], onDelete: Cascade)
  grantee Membership    @relation("AccessGrantsAsGrantee", fields: [granteeId], references: [id], onDelete: Cascade)

  @@index([tenantId])
  @@index([granteeId])
  @@index([patientId])
  @@index([expiresAt])
  @@map("tb_access_grants")
  @@schema("public")
}

import { createNotification, NotificationType } from './notification-store';

// MOCK PROVIDERS FOR DEMO
async function sendEmail(to: string, subject: string, body: string) {
  console.log(`[EMAIL PROVIDER] To: ${to} | Subject: ${subject}`);
}

async function sendSms(to: string, body: string) {
  console.log(`[SMS PROVIDER] To: ${to} | Body: ${body.slice(0, 50)}...`);
}

export type { NotificationType };

export async function notifyPatient(o: {
  tenantId: string;
  patientId: string;
  type: NotificationType;
  title: string;
  message: string;
  resourceType?: string;
  resourceId?: string;
}) {
  // 1. Save to DB (in-app notification)
  const notif = await createNotification(o);
  
  // 2. Fire external channels (mocked for demo)
  // In production, fetch patient email/phone and use Twilio/SendGrid
  sendEmail("patient@example.com", o.title, o.message).catch(() => {});
  sendSms("+254700000000", o.message).catch(() => {});
  
  return notif;
}
